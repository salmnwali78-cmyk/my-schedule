<script>
const STORAGE_KEY = "my_simple_schedule_v1";

const defaultItems = [
{ id: 1, title: "الدروس الصباحية", start: "06:00", end: "12:00", day: "السبت - الأربعاء", done: false },
{ id: 2, title: "فترة المذاكرة", start: "16:00", end: "20:00", day: "السبت - الأربعاء", done: false },
{ id: 3, title: "مراجعة شاملة للأسبوع", start: "16:00", end: "20:00", day: "الخميس", done: false }
];

let items = JSON.parse(localStorage.getItem(STORAGE_KEY)) || defaultItems;

const scheduleList = document.getElementById("schedule-list");
const daySelector = document.getElementById("day-selector");
const form = document.getElementById("settings-form");

function saveItems() {
localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
}

function matchesDay(itemDay, selectedDay) {
if (itemDay === "يومي") return true;
if (itemDay === "السبت - الأربعاء") {
return ["السبت", "الأحد", "الإثنين", "الثلاثاء", "الأربعاء"].includes(selectedDay);
}
return itemDay === selectedDay;
}

function renderSchedule() {
const selectedDay = daySelector.value;
scheduleList.innerHTML = "";

const filtered = items
.filter(item => matchesDay(item.day, selectedDay))
.sort((a, b) => a.start.localeCompare(b.start));

if (filtered.length === 0) {
scheduleList.innerHTML = `<p style="color:#94a3b8; text-align:center;">لا توجد مهام في هذا اليوم. أضف مهمة من الأسفل.</p>`;
return;
}

filtered.forEach(item => {
const div = document.createElement("div");
div.className = `item ${item.done ? "done" : ""}`;
div.innerHTML = `
<div class="item-info">
<input type="checkbox" ${item.done ? "checked" : ""} onchange="toggleDone(${item.id})" />
<div>
<div class="title"><strong>${item.title}</strong></div>
<small style="color:#94a3b8">${item.start} - ${item.end}</small>
</div>
</div>
<div style="display:flex; gap:0.4rem; align-items:center;">
<span class="badge">${item.day}</span>
<button onclick="deleteItem(${item.id})" style="padding:0.35rem 0.6rem; background:#ef4444; border:none; font-size:0.8rem;">حذف</button>
</div>
`;
scheduleList.appendChild(div);
});
}

window.toggleDone = function(id) {
items = items.map(item => item.id === id ? { ...item, done: !item.done } : item);
saveItems();
renderSchedule();
};

window.deleteItem = function(id) {
items = items.filter(item => item.id !== id);
saveItems();
renderSchedule();
};

form.addEventListener("submit", (e) => {
e.preventDefault();
const newItem = {
id: Date.now(),
title: document.getElementById("task-title").value.trim(),
start: document.getElementById("start-time").value,
end: document.getElementById("end-time").value,
day: document.getElementById("task-day").value,
done: false
};
items.push(newItem);
saveItems();
form.reset();
renderSchedule();
});

daySelector.addEventListener("change", renderSchedule);

document.getElementById("export-btn").addEventListener("click", () => {
const blob = new Blob([JSON.stringify(items, null, 2)], { type: "application/json" });
const a = document.createElement("a");
a.href = URL.createObjectURL(blob);
a.download = "جدولي.json";
a.click();
});

document.getElementById("import-input").addEventListener("change", (e) => {
const file = e.target.files[0];
if (!file) return;
const reader = new FileReader();
reader.onload = () => {
try {
items = JSON.parse(reader.result);
saveItems();
renderSchedule();
} catch {
alert("هذا الملف غير صحيح.");
}
};
reader.readAsText(file);
});

document.getElementById("reset-btn").addEventListener("click", () => {
items = [...defaultItems];
saveItems();
renderSchedule();
});

const daysMap = ["الأحد", "الإثنين", "الثلاثاء", "الأربعاء", "الخميس", "الجمعة", "السبت"];
const todayName = daysMap[new Date().getDay()];
daySelector.value = todayName;
document.getElementById("current-day-display").textContent = `اليوم: ${todayName}`;

renderSchedule();
</script>
